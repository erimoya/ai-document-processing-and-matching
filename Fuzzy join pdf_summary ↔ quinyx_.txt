-- Fuzzy join pdf_summary ↔ quinyx_uitzendkrachten_summary
CREATE OR REPLACE TABLE quinyxdata.default.pdf_quinyx_fuzzy_match AS WITH pdf_norm AS (
  SELECT
    datum, weeknummer, jaar, naam, totaal_aantal, totaal,
    bedrijf, factuurnummer,
    array_join(
      array_sort(
        split(regexp_replace(lower(trim(naam)), '[^a-z\\s]', ''), '\\s+')
      ), ' '
    ) AS naam_normalized
  FROM quinyxdata.default.pdf_summary
),
quinyx_weekly AS (
  -- Aggregate quinyx daily rows to week level
  SELECT
    FISCYEAR,
    CC_WEEK,
    NAAM_value,
    SUM(Uren_gewerkt_sum) AS uren_gewerkt_week,
    SUM(Brutoloon_sum)    AS brutoloon_week
  FROM quinyxdata.default.quinyx_uitzendkrachten_summary
  GROUP BY FISCYEAR, CC_WEEK, NAAM_value
),
quinyx_norm AS (
  SELECT
    *,
    array_join(
      array_sort(
        split(regexp_replace(lower(trim(NAAM_value)), '[^a-z\\s]', ''), '\\s+')
      ), ' '
    ) AS naam_normalized
  FROM quinyx_weekly
),
joined AS (
  -- LEFT JOIN on week/year only; name similarity computed per pair
  SELECT
    p.datum,
    p.weeknummer,
    p.jaar,
    p.bedrijf,
    p.factuurnummer,
    p.naam                        AS pdf_naam,
    q.NAAM_value                  AS quinyx_naam,
    ROUND(p.totaal_aantal, 2)     AS pdf_uren,
    ROUND(q.uren_gewerkt_week, 2) AS quinyx_uren,
    ROUND(p.totaal, 2)            AS pdf_totaal,
    ROUND(q.brutoloon_week, 2)    AS quinyx_totaal,
    levenshtein(p.naam_normalized, q.naam_normalized)                  AS edit_distance,
    GREATEST(length(p.naam_normalized), length(q.naam_normalized), 1) AS max_name_len
  FROM pdf_norm p
  LEFT JOIN quinyx_norm q
    ON p.jaar        = CAST(q.FISCYEAR AS INT)
   AND p.weeknummer = CAST(q.CC_WEEK  AS INT)
),
best_match AS (
  -- Pick the single best-matching Quinyx row per PDF row.
  -- NULLS LAST ensures unmatched rows (no Quinyx entry for that week) get rn=1.
  SELECT *,
    ROW_NUMBER() OVER (
      PARTITION BY datum, pdf_naam, factuurnummer
      ORDER BY edit_distance ASC NULLS LAST
    ) AS rn
  FROM joined
),
matched AS (
  -- Keep only the best candidate per PDF row and flag whether it clears the threshold.
  SELECT *,
    CASE
      WHEN edit_distance IS NULL
        OR 1 - (edit_distance / max_name_len) >= 0.8 THEN TRUE
      ELSE FALSE
    END AS is_matched
  FROM best_match
  WHERE rn = 1
)
SELECT
  datum,
  weeknummer,
  jaar,
  bedrijf,
  factuurnummer,
  pdf_naam,
  CASE WHEN is_matched
    THEN COALESCE(quinyx_naam, 'Nog niet geregistreerd bij Quinyx')
    ELSE 'Nog niet geregistreerd bij Quinyx'
  END                                                                        AS quinyx_naam,
  pdf_uren,
  CASE WHEN is_matched THEN quinyx_uren  ELSE NULL END                       AS quinyx_uren,
  ROUND(CASE WHEN is_matched THEN quinyx_uren  ELSE NULL END - pdf_uren, 2) AS verschil_uren,
  pdf_totaal,
  CASE WHEN is_matched THEN quinyx_totaal ELSE NULL END                      AS quinyx_totaal,
  ROUND(CASE WHEN is_matched THEN quinyx_totaal ELSE NULL END - pdf_totaal, 2) AS verschil_totaal
FROM matched
ORDER BY jaar, weeknummer, pdf_naam
