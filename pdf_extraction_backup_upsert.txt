
-- ============================================================
-- STEP 2: Upsert session results into backup
-- Only inserts rows whose business key is not already present.
-- Business key: datum + naam + omschrijving + aantal + tarief + bedrijf + factuurnummer
-- ============================================================
MERGE INTO quinyxdata.default.pdf_extraction_backup AS target
USING (
  SELECT
    sha2(
      concat_ws('|',
        coalesce(cast(datum AS STRING), ''),
        coalesce(naam,         ''),
        coalesce(omschrijving, ''),
        coalesce(cast(aantal   AS STRING), ''),
        coalesce(cast(tarief   AS STRING), ''),
        coalesce(bedrijf,                  ''),
        coalesce(factuurnummer,            '')
      ), 256
    )                      AS record_hash,
    path,
    ocr_text,
    datum,
    weeknummer,
    jaar,
    naam,
    omschrijving,
    aantal,
    percentage,
    tarief,
    subtotaal,
    bedrijf,
    factuurnummer,
    current_user()         AS upload_user,
    current_timestamp()    AS upload_timestamp,
    :session_id            AS session_id
  FROM IDENTIFIER(:pdfMetadataTable)
) AS source
ON target.record_hash = source.record_hash
WHEN NOT MATCHED THEN INSERT *;
