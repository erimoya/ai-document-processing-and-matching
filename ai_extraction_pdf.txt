 -- Create or replace receipt metadata into the final table
CREATE OR REPLACE TABLE IDENTIFIER(:pdfMetadataTable) AS
WITH extracted AS (
  SELECT
    path, -- File path of the pdf
    text AS ocr_text, -- OCR-extracted text
    -- Parse datum from original text via regex (not affected by preprocessing)
    COALESCE(
      -- Timing format: DD-MM-YYYY (e.g. 22-09-2025)
      TRY_TO_DATE(datum_raw, 'dd-MM-yyyy'),
      -- DD/MM/YYYY variant
      TRY_TO_DATE(datum_raw, 'dd/MM/yyyy'),
      -- Olympia format: DD.MM.YYYY (e.g. 23.09.2025)
      TRY_TO_DATE(datum_raw, 'dd.MM.yyyy'),
      -- Covebo format: DD-MM-YY with 2-digit year (e.g. 24-09-25)
      TRY_TO_DATE(datum_raw, 'dd-MM-yy'),
      -- DD/MM/YY variant
      TRY_TO_DATE(datum_raw, 'dd/MM/yy'),
      -- DD.MM.YY variant
      TRY_TO_DATE(datum_raw, 'dd.MM.yy'),
      -- ISO fallback: YYYY-MM-DD
      TRY_CAST(datum_raw AS DATE)
    ) AS datum,
    TRIM(BOTH ' ' FROM SPLIT_PART(CAST(get_json_object(line_item.naam, '$.value') AS STRING), '|', 1)) AS naam,
    get_json_object(line_item.omschrijving, '$.value') AS omschrijving,
    TRY_CAST(get_json_object(line_item.aantal, '$.value') AS DOUBLE) AS aantal,
    TRY_CAST(get_json_object(line_item.percentage, '$.value') AS DOUBLE) AS percentage,
    TRY_CAST(get_json_object(line_item.tarief, '$.value') AS DOUBLE) AS tarief,
    TRY_CAST(get_json_object(line_item.subtotaal, '$.value') AS DOUBLE) AS subtotaal,
    TRY_CAST(get_json_object(line_item.weeknummer, '$.value') AS INT) AS weeknummer,
    bedrijf,
    factuurnummer,
    -- Preserve ordering for window function
    pos AS line_pos
  FROM (
    -- Explode the line_items array into individual rows with position
    SELECT
      path,
      text,
      datum_raw,
      posexplode(
        from_json(
          pdf_metadata:response:line_items::STRING,
          'ARRAY<STRUCT<naam:STRING,omschrijving:STRING,aantal:STRING,percentage:STRING,tarief:STRING,subtotaal:STRING,weeknummer:STRING>>'
        )
      ) AS (pos, line_item),
      pdf_metadata:response:bedrijf:value::STRING       AS bedrijf,
      pdf_metadata:response:factuurnummer:value::STRING AS factuurnummer
    FROM (
      -- Extract pdf metadata using AI
      SELECT
        path, -- File path of the pdf
        text, -- OCR-extracted text from the pdf
        -- Extract date from ORIGINAL text before cleanup using regex
        -- Each PDF vendor uses a different HTML structure for the invoice date:
        --   Timing:  <th>Datum</th>...<td>22-09-2025</td>         (label in <th>, date in <td> of next row)
        --   Olympia: <td>Factuurdatum</td>...<td>23-09-2025</td>  (label in <td>, date in <td> after colspan)
        --   Covebo:  <td>Documentdatum:</td><td>24-09-25</td>     (label in <td> with colon, date directly next)
        COALESCE(
          -- HTML table pattern: handles all three vendors
          -- Matches label (Factuurdatum|Documentdatum|Datum) in either <th> or <td>,
          -- then finds the first <td> containing a date-like value
          NULLIF(regexp_extract(text, '(?si)(?:Factuurdatum|Documentdatum|Datum)[:]?</t[dh]>.*?<td[^>]*>(\\d{1,2}[-/\\.]\\d{1,2}[-/\\.]\\d{2,4})</td>', 1), ''),
          -- Plain text fallback: for PDFs without HTML table structure
          -- Matches label followed by a date within 30 non-digit characters
          NULLIF(regexp_extract(text, '(?i)(?:Factuurdatum|Documentdatum|Datum)[^\\d]{0,30}(\\d{1,2}[-/\\.]\\d{1,2}[-/\\.]\\d{2,4})', 1), '')
        ) AS datum_raw,
        ai_extract(
          -- Pre-process: 5-step cleanup for page-break artifacts
          -- Step 1: strip page-break boilerplate (TE TRANSPORTEREN through GETRANSPORTEERD) - case insensitive
          --         Needed for: Timing (multi-page invoices repeat headers/footers)
          -- Step 2: mark merged cells where next employee name got appended to previous ref (digit<br>Name)
          --         Needed for: Olympia (names merge into previous row's reference cell)
          -- Step 3: remove bank info cells that got merged into data rows
          --         Needed for: Timing (bank payment info bleeds into line item table)
          -- Step 4: merge split tables across Covebo page breaks
          --         Needed for: Covebo (page break splits one table into two with repeated headers)
          --         Removes: </table> + footer/header boilerplate + <table> + header row
          -- Step 5: mark Covebo cells where line item description precedes next employee name
          --         Pattern: <td>Description<br>Name | CAO</td>
          --         → <td>Description<br>[NEXT_EMPLOYEE:]Name | CAO</td>
          --         AI instructions specify: row amounts belong to PREVIOUS employee only
          --         The [NEXT_EMPLOYEE:]'s items come from subsequent rows
          regexp_replace(
            regexp_replace(
              regexp_replace(
                regexp_replace(
                  regexp_replace(text, '(?si)TE TRANSPORTEREN.*?GETRANSPORTEERD', ''),
                  '(?<=\\d\\d\\d)<br>(?=[A-Z][a-z])',
                  '<br>[NEXT_EMPLOYEE:]'
                ),
                '<td[^>]*>\\s*Bank T\\.n\\.v\\.[^<]*(?:<br>[^<]*)*</td>',
                ''
              ),
              '(?s)</table>\\s*\\nSNF.+?COVEBO\\s*<table>\\s*<tr>\\s*<th>.+?</tr>',
              ''
            ),
            '<br>(?=[^<]+ \\| [A-Z][^<]+</td>)',
            '<br>[NEXT_EMPLOYEE:]'
          ),
          '{"line_items": {"type": "array", "description": "All individual line items from the invoice table, one per row", "items": {"type": "object", "properties": {"naam": {"type": "string", "description": "Full name of the employee (e.g. Bodlak, N.B. Nataliia or Akkaya, H. or Hadj Said, H.S.). In Covebo invoices, employee names always contain a pipe symbol (|) followed by a CAO code (e.g. | HANOSHIS_40) - this distinguishes them from line item descriptions. Extract only the name part BEFORE the pipe. The name always appears BEFORE its line items. Propagate the last seen employee name to all subsequent line items until a new name appears."}, "omschrijving": {"type": "string", "description": "Description of the work type or cost (e.g. Normale uren, Uren onregelmatig, Reiskosten onbelast per dag, Koudetoeslag per uur, Loon normaal, Loon onregelm. uren, Bel. Ongemakkentoeslag, Onb.Reiskosten woon/werk)"}, "aantal": {"type": "number", "description": "Number of hours or units worked (e.g. 40.00, 5.00). Use dot as decimal separator. If multiple values are in one cell separated by line breaks, extract each as a separate line item."}, "percentage": {"type": "number", "description": "Percentage surcharge applied (e.g. 100.00, 120.00, 200.00). Null if not applicable."}, "tarief": {"type": "number", "description": "Hourly rate or unit price (e.g. 29.85, 11.40, 30.72, 1.64). Use dot as decimal separator."}, "subtotaal": {"type": "number", "description": "Line total amount (e.g. 1194.00, 1223.58). Use dot as decimal separator. Must approximately equal aantal x tarief (or aantal x percentage/100 x tarief when percentage applies). If a value does NOT match this formula, it is a running page total and must be SKIPPED."}, "weeknummer": {"type": "integer", "description": "The week number for this line item (e.g. 38, 39). Found as a column value (Week or week nr.) in Olympia/Timing invoices, or in the reference cell as Wk:38, or in the document header as Week 2025-38. Extract the numeric week only (e.g. 38). Propagate the last seen week number if not repeated per row."}}}}, "bedrijf": {"type": "string", "description": "Name of the staffing agency (uitzendbureau) that issued the invoice (e.g. Timing, Olympia, Covebo). Usually found in the letterhead or document header."}, "factuurnummer": {"type": "string", "description": "Invoice number or document number (factuurnummer, documentnummer, factuurnr). Usually found in the document header."}}',
          MAP('instructions', 'This is a Dutch staffing agency invoice (uitzendbureau factuur). Extract every line item from the order collection table (Orderverzameling). Each employee may have multiple line items (normal hours, irregular hours, travel costs, cold surcharges). The naam field should contain the employee name from the row above the line items. CRITICAL RULES: 1) Propagate the last seen employee name to all subsequent line items until a new name appears - even across page breaks. 2) When multiple values appear in a single table cell separated by line breaks (<br> tags), each value is a separate line item - extract them individually and match each with its corresponding tarief and subtotaal. 3) The subtotaal must equal approximately aantal x tarief (or aantal x percentage/100 x tarief). If an amount does NOT match this formula, it is a running page total or summary total - skip it entirely. Use this rule to filter valid line items from mixed data. 4) Skip rows that ONLY contain bank info, dates, totals, or administrative data. Do NOT skip a row just because it contains labels like SUBTOT or Getransporteerd - if it also has valid line item data (antal/tarief/subtotaal that pass rule 3), extract those items. 5) IMPORTANT [NEXT_EMPLOYEE:] MARKER: When you see [NEXT_EMPLOYEE:]SomeName anywhere in a cell, the line items in that row (or the description text before the marker in the same cell) belong to the PREVIOUS employee. The name after [NEXT_EMPLOYEE:] is the NEXT employee - their items start from the following row. Two patterns: (A) In a Referentie cell (Timing/Olympia): Wk:38<br>4019384<br>[NEXT_EMPLOYEE:]Hasan, H. with items 38.75/Loon normaal/1190.40 - those items belong to the previous employee (e.g. Georgiev). (B) In a Naam/Omschrijving cell (Covebo): when a cell has text like Kilometervergoeding per dag<br>[NEXT_EMPLOYEE:]Montvidaite Moira | HANOSHIS_40, the text BEFORE the marker (Kilometervergoeding per dag) is the omschrijving and the row amounts (aantal/tarief/subtotaal) belong EXCLUSIVELY to the PREVIOUS employee. CRITICAL: Do NOT create ANY line item for the [NEXT_EMPLOYEE:] name using amounts from THIS row. The [NEXT_EMPLOYEE:] name (Montvidaite Moira) items come ONLY from subsequent rows. Employee names in Covebo always contain " | " followed by a CAO code (e.g. | HANOSHIS_40) - this distinguishes them from descriptions. 6) For bedrijf, extract the full name of the staffing agency from the letterhead or document header. For factuurnummer, extract the invoice number or document number (usually labeled factuurnummer, factuurnr, or documentnummer) from the document header. 7) For weeknummer, extract the numeric week number from the table column (Week, week nr.), from reference cells (Wk:38), or from the document header (Week 2025-38). Propagate the last seen week number to all line items until a new week appears.')
        ) AS pdf_metadata -- Struct containing extracted metadata fields
      FROM IDENTIFIER(:ocrTable)
      WHERE error_status IS NULL -- Only process pdfs without errors
      LIMIT INT(:limit)
    ) AS ai_data
  ) AS exploded_data
)
-- Post-processing: fill any remaining NULL names with the last known name per document
SELECT
  path,
  ocr_text,
  datum,
  weeknummer,
  YEAR(datum) AS jaar,
  COALESCE(
    naam,
    LAST_VALUE(naam) IGNORE NULLS OVER (
      PARTITION BY path
      ORDER BY line_pos
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    )
  ) AS naam,
  bedrijf,
  factuurnummer,
  omschrijving,
  aantal,
  percentage,
  tarief,
  subtotaal
FROM extracted
